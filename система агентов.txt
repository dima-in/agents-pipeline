Система многоагентной разработки с контролем качества
Отличная идея! Давайте структурируем вашу систему.

QA расшифровка
QA = Quality Assurance (Обеспечение качества)

Предлагаемая архитектура
1. Цикл разработки (Development Loop)
code
Копировать
Architect → Developer → QA → [успех/откат]
Автоматика через Git:

Каждый агент работает в своей ветке
QA проверяет изменения и либо мерджит, либо откатывает
При откате QA добавляет комментарии в задачу

    def _log_json(self, event_type, data):
        """Логирование в JSON формат"""
        event = {
            "timestamp": datetime.now().isoformat(),
            "type": event_type,
            "data": data
        }
        self.json_events.append(event)
        
        # Сохраняем в файл
        with open(self.json_log, 'w', encoding='utf-8') as f:
            json.dump(self.json_events, f, indent=2, ensure_ascii=False)
    
    def save_summary(self):
        """Сохранение итогового отчета"""
        summary_file = self.log_dir / f"summary_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
        
        with open(summary_file, 'w', encoding='utf-8') as f:
            f.write("=" * 60 + "\n")
            f.write("ИТОГОВЫЙ ОТЧЕТ WORKFLOW\n")
            f.write("=" * 60 + "\n\n")
            
            phases = [e for e in self.json_events if e['type'] == 'phase_start']
            agents = [e for e in self.json_events if e['type'] == 'agent_end']

# workflow/orchestrator.py
import subprocess
import time
import yaml
from pathlib import Path
from logger import WorkflowLogger
import git
from typing import Dict, List, Optional

class WorkflowOrchestrator:
    def __init__(self, config_path="workflow/config.yaml"):
        self.config = self._load_config(config_path)
        self.logger = WorkflowLogger()
        self.repo = git.Repo('.')
        self.current_branch = None
        self.task_counter = 0
        
    def _load_config(self, config_path):
        """Загрузка конфигурации"""
        with open(config_path, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f)
    
    def _wait_for_user(self, prompt="Продолжить?", auto_mode=False):
        """Ожидание подтверждения пользователя"""
        if auto_mode or self.config['workflow']['mode'] == 'auto':
            delay = self.config['workflow']['auto_continue_delay']
            self.logger.info(f"Авто-продолжение через {delay} секунд...")
            time.sleep(delay)
            return True
        
        self.logger.user_input_required(prompt)
        while True:
            response = input(f"  [{prompt}] (y/n/auto): ").lower().strip()
            if response == 'y':
                return True
            elif response == 'n':
                return False
            elif response == 'auto':
                self.config['workflow']['mode'] = 'auto'
                return True
            else:
                print("  Введите 'y' (да), 'n' (нет) или 'auto' (автоматически)")
    
    def _run_agent(self, agent_config: Dict, phase: str) -> bool:
        """Запуск отдельного агента"""
        agent_name = agent_config['name']
        description = agent_config.get('description', '')
        timeout = agent_config.get('timeout', 600)
        
        self.logger.agent_start(agent_name, description)
        
        try:
            # Команда запуска агента
            agent_dir = f".openclaw/agents/{phase}/{agent_name}"
            cmd = [
                "openclaw", "run", agent_name,
                "--agent-dir", agent_dir,
                "--workspace", "."
            ]
            
            self.logger.agent_progress(agent_name, f"Выполнение команды: {' '.join(cmd)}")
            
            # Запуск процесса
            process = subprocess.Popen(
                cmd,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                text=True,
                encoding='utf-8'
            )
            
            # Читаем вывод в реальном времени
            for line in process.stdout:
                self.logger.agent_progress(agent_name, line.strip())
            
            # Ожидаем завершения
            return_code = process.wait(timeout=timeout)
            
            if return_code == 0:
                self.logger.agent_end(agent_name, "success", "Задача выполнена успешно")
                return True
            else:
                stderr = process.stderr.read()
                self.logger.agent_end(agent_name, "failed", stderr)
                return False
                
        except subprocess.TimeoutExpired:
            self.logger.error(f"Агент {agent_name} превысил таймаут {timeout}с")
            process.kill()
            self.logger.agent_end(agent_name, "failed", "Timeout")
            return False
        except Exception as e:
            self.logger.error(f"Ошибка при запуске агента {agent_name}", str(e))
            self.logger.agent_end(agent_name, "failed", str(e))
            return False

            cwd=self.project_root,
            capture_output=True,
            text=True,
            timeout=300  # 5 минут максимум
        )

        log_files = list(log_dir.glob("workflow_*.log"))
        self.assertGreater(len(log_files), 0, "No log files created")
        
        # Проверяем что созданы JSON логи
        json_logs = list(log_dir.glob("workflow_*.json"))
        self.assertGreater(len(json_logs), 0, "No JSON logs created")
        
        # Читаем последний JSON лог
        latest_json = max(json_logs, key=lambda p: p.stat().st_mtime)
        with open(latest_json, 'r') as f:

        # Проверяем структуру лога
